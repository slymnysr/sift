terraform {
  required_version = ">= 1.6"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.bolge
}

variable "bolge" {
  type        = string
  default     = "eu-central-1"
  description = "Kaynaklarin acilacagi bolge"
}

variable "ortam" {
  type = string
}

resource "aws_s3_bucket" "kayitlar" {
  bucket = "kayitlar-${var.ortam}"

  lifecycle {
    prevent_destroy = true
  }
}

resource "aws_s3_bucket_versioning" "kayitlar" {
  bucket = aws_s3_bucket.kayitlar.id

  versioning_configuration {
    status = "Enabled"
  }
}

output "kayit_kovasi" {
  value = aws_s3_bucket.kayitlar.arn
}
